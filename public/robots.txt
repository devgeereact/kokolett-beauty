# The owner's dashboard, a customer's own bookings, and the single-use
# sign-in links. None of these should ever appear in a search result: an
# indexed /access/<token> is a customer's booking history handed to whoever
# searches for it.
Disallow: /dashboard
Disallow: /my
# Trailing slash on purpose: robots.txt matching is a prefix match, so a bare
# "/access" would also hide /accessibility, which is a public page.
Disallow: /access/
Disallow: /login
# Single-use links from an email. Neither is a page anyone should reach from a
# search result, and an indexed /unsubscribe/<id> lets a stranger unsubscribe
# someone else.
Disallow: /reset-password
Disallow: /unsubscribe
